vad-project/
│
├── data/                       # Dati in input e output (Da NON caricare su Git)
│   ├── raw_videos/             # I video registrati con il visore (es. .mp4).
│   └── extracted_frames/       # I singoli frame estratti dai video per testare il VLM.
│
├── models/                     # Modelli scaricati in locale (Da NON caricare su Git)
│   ├── bonsai_27b_1bit/        # Qui salverai i pesi quantizzati di Bonsai.
│   └── c_radio_weights/        # Eventuali pesi o checkpoint specifici di C-RADIOv4.
│
├── configs/                    # Configurazioni statiche e dizionari (Fase 0)
│   ├── visual_dictionary.txt   # Il prompt descrittivo per C-RADIOv4.
│   └── dependency_graph.json   # Le regole logiche per l'LLM (lo useremo più avanti).
│
├── src/                        # Il codice sorgente del progetto
│   ├── c_radio/                # Qui clonerai la repository ufficiale di NVIDIA.
│   ├── perception/             # Script per la Fase 1 (visione).
│   │   ├── extract_frames.py   # Script per estrarre i frame dai video in 'raw_videos/'.
│   │   └── object_grounding.py # Lo script che invia i frame e il dizionario a C-RADIOv4.
│   └── utils/                  # Funzioni di supporto (es. per disegnare le bounding box sui frame).
│
├── notebooks/                  # Jupyter Notebooks per sperimentazioni rapide
│   └── phase1_grounding_test.ipynb # Ideale per caricare un frame, far girare il VLM e visualizzare subito la maschera a schermo.
│
├── logs/                       # File di log generati automaticamente dagli script
│   └── vlm_outputs/            # Per salvare i JSON o i log testuali di cosa vede C-RADIOv4.
│
├── .gitignore                  # Fondamentale: dirà a Git di ignorare data/, models/, logs/, venv/ e __pycache__/.
├── requirements.txt            # La lista delle dipendenze Python (torch, mlx, opencv, ecc.).
├── README.md                   # Descrizione del progetto, istruzioni di installazione e avvio.
└── trials_and_errors_log.md    # Il diario di bordo per tracciare ogni esperimento, fallimento e soluzione adottata.
